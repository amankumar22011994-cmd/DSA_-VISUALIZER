// ━━━━ Bubble Sort ━━━━
void bubbleSort(vector<int>& a) {
    int n = a.size();
    for(int i=0;i<n-1;i++)
        for(int j=0;j<n-i-1;j++)
            if(a[j]>a[j+1]) swap(a[j],a[j+1]);
}

// ━━━━ Merge Sort ━━━━
void merge(vector<int>& a, int l, int m, int r) {
    vector<int> tmp;
    int i=l, j=m+1;
    while(i<=m && j<=r)
        tmp.push_back(a[i]<=a[j] ? a[i++] : a[j++]);
    while(i<=m) tmp.push_back(a[i++]);
    while(j<=r) tmp.push_back(a[j++]);
    for(int k=l;k<=r;k++) a[k]=tmp[k-l];
}
void mergeSort(vector<int>& a, int l, int r) {
    if(l<r) {
        int m=(l+r)/2;
        mergeSort(a,l,m); mergeSort(a,m+1,r);
        merge(a,l,m,r);
    }
}

// ━━━━ STANDARD BINARY SEARCH ━━━━
int binarySearch(vector<int>& arr, int target) {
    int lo=0, hi=arr.size()-1;
    while(lo <= hi) {
        int mid = lo + (hi-lo)/2; // NO overflow! (not (lo+hi)/2)
        if(arr[mid] == target) return mid;
        if(arr[mid] < target)  lo = mid+1;
        else                    hi = mid-1;
    }
    return -1; // not found
}

// ━━━━ FIND FIRST OCCURRENCE ━━━━
int firstOcc(vector<int>& arr, int x) {
    int lo=0, hi=arr.size()-1, ans=-1;
    while(lo<=hi) {
        int mid=lo+(hi-lo)/2;
        if(arr[mid]==x){ ans=mid; hi=mid-1; } // ans record, left search
        else if(arr[mid]<x) lo=mid+1;
        else hi=mid-1;
    }
    return ans;
}

// ━━━━ BINARY SEARCH ON ANSWER (powerful pattern!) ━━━━
// "minimum/maximum value dhundho satisfying condition"
int lo=1, hi=1e9, ans=-1;
while(lo<=hi) {
    int mid=lo+(hi-lo)/2;
    if(canAchieve(mid)) { ans=mid; hi=mid-1; } // minimize
    else lo=mid+1;
}
⚠️
Classic Bug: mid = (lo+hi)/2 overflow ho sakta hai jab dono bade hain! Hamesha mid = lo + (hi-lo)/2 likho.

Kadane's Algorithm

Maximum subarray sum — O(n) DP. dp[i] = max subarray ending at i

int maxSubArray(vector<int>& a) {
    int cur=a[0], best=a[0];
    for(int i=1;i<a.size();i++){
        cur = max(a[i], cur+a[i]);
        best = max(best, cur);
    }
    return best;
}

Dutch National Flag (3-way partition)

0s, 1s, 2s sort karo in O(n) — 3 pointers!

void sort012(vector<int>& a) {
    int lo=0, mid=0, hi=a.size()-1;
    while(mid<=hi){
        if(a[mid]==0)      swap(a[lo++],a[mid++]);
        else if(a[mid]==1) mid++;
        else               swap(a[mid],a[hi--]);
    }
}

Cyclic Sort

1 to N range ke numbers — har number apni sahi jagah pe rakho O(n)

void cyclicSort(vector<int>& a) {
    int i=0;
    while(i<a.size()){
        int j=a[i]-1; // correct position
        if(a[i]!=a[j]) swap(a[i],a[j]);
        else i++;
    }
}
