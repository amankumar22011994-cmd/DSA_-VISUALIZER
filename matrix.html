// ━━━━ SEARCH IN ROW+COL SORTED MATRIX O(n+m) ━━━━
// Start from top-right corner
bool searchMatrix(vector<vector<int>>& mat, int target) {
    int n=mat.size(), m=mat[0].size();
    int r=0, c=m-1;  // top-right
    while(r<n && c>=0) {
        if(mat[r][c]==target) return true;
        if(mat[r][c]<target) r++;   // too small → go down
        else                 c--;   // too big  → go left
    }
    return false;
}

// ━━━━ LEETCODE 74: SEARCH (rows sorted, first>last of prev) ━━━━

bool searchMatrix74(vector<vector<int>>& mat, int target) {
    int n=mat.size(), m=mat[0].size();
    int lo=0, hi=n*m-1;
    while(lo<=hi) {
        int mid=lo+(hi-lo)/2;
        int val=mat[mid/m][mid%m]; // 1D → 2D index!
        if(val==target) return true;
        if(val<target)  lo=mid+1;
        else            hi=mid-1;
    }
    return false;
}

// ━━━━ DFS TEMPLATE (Island / Flood Fill) ━━━━
int dr[] = {-1,1,0,0};
int dc[] = {0,0,-1,1};

void dfs(vector<vector<int>>& grid, int r, int c) {
    int n=grid.size(), m=grid[0].size();
    if(r<0||r>=n||c<0||c>=m) return; // boundary
    if(grid[r][c]!=1) return;         // not land/visited
    grid[r][c] = 2;                    // mark visited
    for(int d=0;d<4;d++)
        dfs(grid, r+dr[d], c+dc[d]);
}

int numIslands(vector<vector<char>>& grid) {
    int count=0;
    for(int i=0;i<grid.size();i++)
        for(int j=0;j<grid[0].size();j++)
            if(grid[i][j]=='1'){ dfs(grid,i,j); count++; }
    return count;
}

// ━━━━ BFS TEMPLATE (Shortest Path / Flood) ━━━━
int bfsShortestPath(vector<vector<int>>& grid, pair<int,int> src, pair<int,int> dst) {
    int n=grid.size(), m=grid[0].size();
    vector<vector<bool>> vis(n,vector<bool>(m,false));
    queue<pair<int,int>> q;
    q.push(src); vis[src.first][src.second]=true;
    int dist=0;
    while(!q.empty()) {
        int sz=q.size();
        while(sz--) {
            auto [r,c]=q.front(); q.pop();
            if(make_pair(r,c)==dst) return dist;
            for(int d=0;d<4;d++){
                int nr=r+dr[d],nc=c+dc[d];
                if(nr>=0&&nr<n&&nc>=0&&nc<m&&!vis[nr][nc]&&grid[nr][nc]==0){
                    vis[nr][nc]=true; q.push({nr,nc});
                }
            }
        }
        dist++;
    }
    return -1;
}

2D Prefix Sum Code
// ━━━━ BUILD 2D PREFIX SUM ━━━━
int n=4, m=4;
vector<vector<int>> pre(n+1, vector<int>(m+1, 0));

for(int i=1;i<=n;i++)
    for(int j=1;j<=m;j++)
        pre[i][j] = mat[i-1][j-1]
                  + pre[i-1][j]    // top
                  + pre[i][j-1]    // left
                  - pre[i-1][j-1]; // subtract overlap

// ━━━━ QUERY: sum of rectangle (r1,c1) to (r2,c2) ━━━━
int query(int r1, int c1, int r2, int c2) {
    return pre[r2+1][c2+1]
         - pre[r1][c2+1]    // subtract top
         - pre[r2+1][c1]    // subtract left
         + pre[r1][c1];       // add back corner
}

Spiral Matrix
MatrixSimulation
m×n matrix ka elements spiral order mein return karo.

4 boundaries rakhte hain: top, bottom, left, right. Clockwise: left→right (top++), top→bottom (right--), right→left (bottom--), bottom→top (left++). Jab boundaries cross ho jaayein, stop.

vector<int> spiralOrder(vector<vector<int>>& mat) {
    vector<int> res;
    if(mat.empty()) return res;
    int top=0,bot=mat.size()-1,left=0,right=mat[0].size()-1;
    while(top<=bot && left<=right) {
        for(int i=left;i<=right;i++) res.push_back(mat[top][i]); top++;
        for(int i=top;i<=bot;i++) res.push_back(mat[i][right]); right--;
        if(top<=bot){for(int i=right;i>=left;i--) res.push_back(mat[bot][i]); bot--;}
        if(left<=right){for(int i=bot;i>=top;i--) res.push_back(mat[i][left]); left++;}
    }
    return res;
}
Time: O(n×m)Space: O(1)
#48Medium
Rotate Image
MatrixIn-place
n×n matrix ko 90° clockwise rotate karo in-place (no extra matrix).

Step 1: Transpose — mat[i][j] aur mat[j][i] swap karo (j=i+1 se start, duplicate avoid karo).
Step 2: Reverse each row — har row ko reverse karo. Done! No extra space needed.

void rotate(vector<vector<int>>& mat) {
    int n = mat.size();
    // Step 1: Transpose
    for(int i=0;i<n;i++)
        for(int j=i+1;j<n;j++)
            swap(mat[i][j], mat[j][i]);
    // Step 2: Reverse each row
    for(int i=0;i<n;i++)
        reverse(mat[i].begin(), mat[i].end());
}
Time: O(n²)Space: O(1) in-place

Set Matrix Zeroes
MatrixIn-place
Agar kisi cell mein 0 hai, toh uski poori row aur column 0 kar do. In-place O(1) space.

First row aur first column ko markers ki tarah use karo. Pehle check karo ki first row/col mein 0 hai ya nahi (alag flag). Phir baaki matrix scan karo — jahan 0 mile, mat[i][0] aur mat[0][j] = 0 mark karo. Phir markers ke hisaab se rows/cols zero karo.

void setZeroes(vector<vector<int>>& mat) {
    int n=mat.size(), m=mat[0].size();
    bool row0=false, col0=false;
    // Check first row & col
    for(int j=0;j<m;j++) if(!mat[0][j]) row0=true;
    for(int i=0;i<n;i++) if(!mat[i][0]) col0=true;
    // Mark rows & cols using first row/col
    for(int i=1;i<n;i++) for(int j=1;j<m;j++)
        if(!mat[i][j]) mat[i][0]=mat[0][j]=0;
    // Zero out marked rows & cols
    for(int i=1;i<n;i++) if(!mat[i][0]) for(int j=1;j<m;j++) mat[i][j]=0;
    for(int j=1;j<m;j++) if(!mat[0][j]) for(int i=1;i<n;i++) mat[i][j]=0;
    if(row0) for(int j=0;j<m;j++) mat[0][j]=0;
    if(col0) for(int i=0;i<n;i++) mat[i][0]=0;
}
Time: O(n×m)Space: O(1)

bool searchMatrix(vector<vector<int>>& mat, int target) {
    int n=mat.size(), m=mat[0].size();
    int lo=0, hi=n*m-1;
    while(lo<=hi) {
        int mid=lo+(hi-lo)/2;

Top-right corner se start karo. Agar value == target → found. Agar value < target → row badhao (neeche jao). Agar value > target → col ghataao (left jao). O(n+m) guaranteed!

bool searchMatrix(vector<vector<int>>& mat, int target) {

    while(r<n && c>=0) {
        if(mat[r][c]==target) return true;
        mat[r][c]<target ? r++ : c--;

Range Sum Query 2D - Immutable
2D Prefix SumDesign
Multiple range sum queries efficiently answer karo — submatrix sum O(1) mein.

2D prefix sum banao. Query = pre[r2+1][c2+1] - pre[r1][c2+1] - pre[r2+1][c1] + pre[r1][c1]. O(n×m) build, O(1) query.

class NumMatrix {
    vector<vector<int>> pre;
public:
    NumMatrix(vector<vector<int>>& mat) {
        int n=mat.size(), m=mat[0].size();
        pre.assign(n+1, vector<int>(m+1,0));
        for(int i=1;i<=n;i++) for(int j=1;j<=m;j++)
            pre[i][j]=mat[i-1][j-1]+pre[i-1][j]+pre[i][j-1]-pre[i-1][j-1];
    }
    int sumRegion(int r1,int c1,int r2,int c2) {
        return pre[r2+1][c2+1]-pre[r1][c2+1]-pre[r2+1][c1]+pre[r1][c1];
    }
};

Rotting Oranges
BFS Multi-sourceMatrix
Grid mein 0=empty, 1=fresh, 2=rotten orange. Har minute rotten adjacent ko rot karta hai. Minutes return karo.

Multi-source BFS — sab rotten oranges simultaneously queue mein daalo. Level by level BFS = minutes. Fresh count track karo — agar end mein fresh > 0, return -1.

int orangesRotting(vector<vector<int>>& g) {
    int n=g.size(),m=g[0].size(),fresh=0,mins=0;
    queue<pair<int,int>> q;
    for(int i=0;i<n;i++) for(int j=0;j<m;j++) {
        if(g[i][j]==2) q.push({i,j});
        if(g[i][j]==1) fresh++;
    }
    int dr[]={-1,1,0,0}, dc[]={0,0,-1,1};
    while(!q.empty() && fresh) {
        int sz=q.size(); mins++;
        while(sz--) {
            auto[r,c]=q.front();q.pop();
            for(int d=0;d<4;d++){
                int nr=r+dr[d],nc=c+dc[d];
                if(nr>=0&&nr<n&&nc>=0&&nc<m&&g[nr][nc]==1){
                    g[nr][nc]=2;fresh--;q.push({nr,nc});
                }
            }
        }
    }
    return fresh?-1:mins;
}

            int nr=r+dr[d],nc=c+dc[d];
            if(nr>=0&&nr<n&&nc>=0&&nc<m&&dist[nr][nc]>dist[r][c]+1){
                dist[nr][nc]=dist[r][c]+1;q.push({nr,nc});
            }
        }
    }
    return dist;
}
