XOR Operator ^ ★ Most Important!

Dono bits alag hoti hain tabhi 1. Same = 0, Different = 1. XOR ka magic property: a^a=0, a^0=a

// XOR (^) — sabse versatile operator!
int a = 12;  // 1100
int b = 10;  // 1010
cout << (a ^ b); // 0110 = 6

// XOR Magic Properties:
// a ^ a = 0     (same number XOR = 0)
// a ^ 0 = a     (XOR with 0 = same)
// a ^ b ^ a = b (cancel ho jaata hai)

// ★ Swap without temp variable
a ^= b; b ^= a; a ^= b;

// ★ Toggle kth bit
n ^= (1 << k);

// ★ Find single number in array
int result = 0;
for(int x : arr) result ^= x;

Bit Check/Set/Clear/Toggle — Core 4 Operations
int n = 42, k = 3; // k = bit position (0-indexed)

// ★ 1. Check kth bit (0 ya 1?)
bool isSet = (n >> k) & 1;
//           OR
bool isSet2 = n & (1 << k);

// ★ 2. Set kth bit (1 karo)
n |= (1 << k);

// ★ 3. Clear kth bit (0 karo)
n &= ~(1 << k);

// ★ 4. Toggle kth bit (flip karo)
n ^= (1 << k);
// Even/Odd Check
Last bit 1 = odd, 0 = even. Division se 2x fast.
// if(n%2) → slow
if(n & 1) cout<<"Odd";
else      cout<<"Even";
// Power of 2 Check
Power of 2 ke binary mein sirf 1 bit hoti hai.
// 8 = 1000, 7 = 0111
// 8 & 7 = 0 → power of 2
bool isPow2 = n&&!(n&(n-1));
// Swap Without Temp
XOR property use karke swap — extra variable nahi.

a ^= b;
// a aur b swap ho gaye!
// Rightmost Set Bit
n & (-n) rightmost 1 bit isolate karta hai.

// 12 (1100) → 4 (0100)
// 10 (1010) → 2 (0010)
// Remove Rightmost Set Bit
n & (n-1) se rightmost 1 bit remove hoti hai. Popcount mein kaam aata hai.

// Count Set Bits (Popcount)
Kitne 1 bits hain? 3 tarike hain.
// Method 1: Built-in (fastest)

// Method 2: n&(n-1) trick
int cnt=0;
while(n){n&=(n-1);cnt++;}

// Method 3: Bit shifting

Shifts division/multiplication se fast hain.

int div = n >> k; // n / 2^k

// n * 8 = n << 3
// n / 4 = n >> 2
// Upper/Lower Case Toggle
ASCII difference sirf ek bit ka hai!
// 'A'=65 (1000001)
// 'a'=97 (1100001) ← bit 5
c | 32   // to lowercase
c & ~32  // to uppercase
c ^ 32   // toggle case
GCC Built-in Functions — CP Essential
// __builtin_popcount(n)  → count of set bits (1s)
__builtin_popcount(13);  // 3 (1101 → three 1s)

// __builtin_clz(n)       → count leading zeros
__builtin_clz(8);        // 28 (for 32-bit int)

// __builtin_ctz(n)       → count trailing zeros
__builtin_ctz(12);       // 2 (1100 → 2 trailing zeros)

// __builtin_parity(n)    → 1 if odd set bits, 0 if even
__builtin_parity(7);     // 1 (111 → 3 ones = odd)

// For long long: use __builtin_popcountll()
__builtin_popcountll(1LL<<40);

Single Number — XOR trick
// Sabhi XOR karo — pairs cancel hote hain
// a^a=0, a^0=a
int singleNumber(vector<int>& nums) {
    int res = 0;
    for(int x : nums) res ^= x;
    return res;
}
// [1,1,2,2,3] → 3 ✓

Missing Number (1 to N)
// 0^1^..^N XOR arr elements
int missingNumber(vector<int>& nums) {
    int n = nums.size(), res = n;
    for(int i=0; i<n; i++)
        res ^= i ^ nums[i];
    return res;
}
// [3,0,1] → 2 ✓

Count Set Bits (Hamming Weight)
int hammingWeight(int n) {
    int count = 0;
    while(n) {
        n &= (n-1); // rightmost bit clear
        count++;
    }
    return count;
    // ya: return __builtin_popcount(n);
}

Power of 2 Check
bool isPowerOfTwo(int n) {
    // Power of 2: sirf 1 bit set
    // n & (n-1) removes that bit → 0
    return n > 0 && !(n & (n-1));
}
// 8=1000, 7=0111, 8&7=0 → true
// 6=0110, 5=0101, 6&5=4 → false

Reverse Bits
uint32_t reverseBits(uint32_t n) {
    uint32_t res = 0;
    for(int i=0;i<32;i++) {
        res = (res<<1) | (n&1);
        n >>= 1;
    }
    return res;
}

Two's Complement — Negative Numbers
// Positive numbers: direct binary
// 5  = 00000101

// Negative: Two's complement = ~n + 1
// -5 = ~5 + 1 = 11111010 + 1 = 11111011

// Why useful?
// -n = ~n + 1  (negate karna)
// n & (-n) = rightmost set bit (wahi use hota hai!)

int n = 12; // 1100
cout << (n & -n); // 4 (0100) — rightmost 1

// Signed vs Unsigned shift:
// signed >>  : arithmetic (sign extend)
// unsigned >>: logical (fill with 0)
int neg = -8;
cout << (neg >> 1); // -4 (sign preserved)
unsigned int u = (unsigned int)neg;
cout << (u >> 1);   // large positive
Bitmask DP — Advanced CP
// TSP (Travelling Salesman) — Bitmask DP
// dp[mask][i] = min cost to visit all nodes in mask, ending at i
int n = 4;
vector<vector<int>> dp(1<<n, vector<int>(n, 1e9));
dp[1][0] = 0; // start at node 0

for(int mask=1; mask<(1<<n); mask++) {
    for(int u=0; u<n; u++) {
        if(!(mask & (1<<u))) continue;
        for(int v=0; v<n; v++) {
            if(mask & (1<<v)) continue;
            int newMask = mask | (1<<v);
            dp[newMask][v] = min(dp[newMask][v],
                                 dp[mask][u] + dist[u][v]);
        }
    }
}
Fenwick Tree (BIT) — Binary Indexed Tree
// BIT uses bit manipulation for range queries
// i & (-i) = lowest set bit (key operation!)
vector<int> bit(n+1, 0);

void update(int i, int val) {
    for(; i <= n; i += i & (-i)) // next index
        bit[i] += val;
}

int query(int i) {
    int sum = 0;
    for(; i > 0; i -= i & (-i)) // parent index
        sum += bit[i];
    return sum;
}
// Prefix sum in O(log n)!
XOR Basis / Linear Basis — Competitive Programming
// XOR basis: set of numbers se max XOR banao
int basis[32] = {};

void insert(int x) {
    for(int i=31; i>=0; i--) {
        if(!(x >> i)) continue;
        if(!basis[i]) { basis[i]=x; break; }
        x ^= basis[i];
    }
}

int maxXOR() {
    int res = 0;
    for(int i=31; i>=0; i--)
        res = max(res, res ^ basis[i]);
    return res;
}
